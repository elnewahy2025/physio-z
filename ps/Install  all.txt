import { Injectable, NotFoundException } from '@nestjs/common';
import * as fs from 'fs';
import * as path from 'path';

/**

File storage service with local implementation
Designed to be easily replaceable with cloud storage (S3, etc.)
*/
@Injectable()
export class FileStorageService {
private readonly storageDir: string;
private readonly isLocal: boolean;
constructor() {
// For now, always use local storage
// Later, this can be configured via environment
this.storageDir = process.env.UPLOAD_DIR || './uploads';
this.isLocal = true;

// Ensure storage directory exists
if (!fs.existsSync(this.storageDir)) {
fs.mkdirSync(this.storageDir, { recursive: true });
}
}

/**

Upload a file
@param file - The file to upload
@param subdirectory - Subdirectory (e.g., 'patients', 'exercises')
@returns File information including URL
*/
async upload(
file: Express.Multer.File,
subdirectory: string = 'general'
): Promise<{ url: string; filename: string; size: number; mimeType: string }> {
const filename = this.generateFilename(file.originalname);
const fullPath = path.join(this.storageDir, subdirectory);
// Create subdirectory if needed
if (!fs.existsSync(fullPath)) {
fs.mkdirSync(fullPath, { recursive: true });
}

const filePath = path.join(fullPath, filename);

if (this.isLocal) {
// Local storage
fs.writeFileSync(filePath, file.buffer);

return {
url: `/uploads/${subdirectory}/${filename}`,
filename: file.originalname,
size: file.size,
mimeType: file.mimetype,
};
}

// Future: Cloud storage (S3, Cloudflare R2, etc.)
// This is where cloud upload logic would go
throw new Error('Cloud storage not yet implemented');
}

/**

Download/Retrieve a file
@param fileUrl - The file URL (relative path)
@returns File buffer and metadata
*/
async retrieve(fileUrl: string): Promise<{ buffer: Buffer; mimeType: string }> {
// Remove leading slash for path.join
const relativePath = fileUrl.replace(/^//, '');
const fullPath = path.join(process.cwd(), relativePath);
if (!fs.existsSync(fullPath)) {
throw new NotFoundException(`File not found: ${fileUrl}`);
}

const buffer = fs.readFileSync(fullPath);
const mimeType = this.getMimeType(path.extname(fullPath));

return { buffer, mimeType };
}

/**

Delete a file
@param fileUrl - The file URL to delete
*/
async delete(fileUrl: string): Promise<void> {
const relativePath = fileUrl.replace(/^//, '');
const fullPath = path.join(process.cwd(), relativePath);
if (fs.existsSync(fullPath)) {
fs.unlinkSync(fullPath);
}
}

/**

Check if file exists
@param fileUrl - The file URL to check
*/
async exists(fileUrl: string): Promise<boolean> {
const relativePath = fileUrl.replace(/^//, '');
const fullPath = path.join(process.cwd(), relativePath);
return fs.existsSync(fullPath);
}
/**

Get storage statistics
*/
async getStats(): Promise<{ totalFiles: number; totalSize: number; freeSpace: number }> {
const files = this.getAllFiles(this.storageDir);
let totalSize = 0;
for (const file of files) {
const stats = fs.statSync(file);
totalSize += stats.size;
}

return {
totalFiles: files.length,
totalSize,
freeSpace: fs.statfsSync(this.storageDir).bavail * fs.statfsSync(this.storageDir).bsize,
};
}

/**

Generate unique filename
*/
private generateFilename(originalName: string): string {
const extension = path.extname(originalName);
const timestamp = Date.now();
const random = uuidv4().substring(0, 8);
return ${timestamp}_${random}${extension};
}
/**

Get MIME type from extension
*/
private getMimeType(extension: string): string {
const mimeTypes: Record<string, string> = {
'.jpg': 'image/jpeg',
'.jpeg': 'image/jpeg',
'.png': 'image/png',
'.gif': 'image/gif',
'.pdf': 'application/pdf',
'.doc': 'application/msword',
'.docx': 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
};
return mimeTypes[extension.toLowerCase()] || 'application/octet-stream';
}

/**

Recursively get all files in directory
*/
private getAllFiles(dir: string): string[] {
let files: string[] = [];
const items = fs.readdirSync(dir);
for (const item of items) {
const fullPath = path.join(dir, item);
const stat = fs.statSync(fullPath);

if (stat.isDirectory()) {
files = files.concat(this.getAllFiles(fullPath));
} else {
files.push(fullPath);
}
}

return files;
}

CRITICAL SECURITY SERVICE:
Ensures that patients can only access their own data,
and that access rules are enforced based on user roles.
Security Rules:
PATIENT: Can only access their own data
THERAPIST: Can access patients they have treated
SECRETARY: Can access all patient data (read-only typically)
OWNER: Full access to all data
*/
@Injectable()
export class PatientDataIsolationService {
constructor(private prisma: PrismaService) {}
/**

Verify that the current user can access the specified patient's data
@param patientId - The patient ID being accessed
@param userId - The current user's ID
@param userRole - The current user's role
@throws ForbiddenException if access is denied
@example
typescript

Enter your code here...
// In your controller/service:
await this.dataIsolation.verifyPatientAccess(
patientId,
req.user.id,
req.user.role
);
text

Enter your code here...
*/
async verifyPatientAccess(
patientId: string,
userId: string,
userRole: string
): Promise<void> {
// OWNER and SECRETARY can access all patient data
if (userRole === 'OWNER' || userRole === 'SECRETARY') {
return;
}

// THERAPIST can access their assigned/current patients
if (userRole === 'THERAPIST') {
// Check if therapist has any appointment with this patient
const appointment = await this.prisma.appointment.findFirst({
where: {
patientId,
therapistId: userId,
},
});

if (appointment) {
return;
}

// Check if therapist has any past appointments with this patient
const pastAppointment = await this.prisma.appointment.findFirst({
where: {
patientId,
therapistId: userId,
dateTime: {
lt: new Date(),
},
},
});

if (pastAppointment) {
return;
}

throw new ForbiddenException(
'يمكنك فقط الوصول إلى المرضى الذين عالجتهم' // You can only access patients you have treated
);
}

// PATIENT can only access their own data
if (userRole === 'PATIENT') {
const patient = await this.prisma.patient.findFirst({
where: {
id: patientId,
userId,
},
});

if (patient) {
return;
}

throw new ForbiddenException(
'يمكنك فقط الوصول إلى بياناتك الخاصة' // You can only access your own data
);
}

// Default deny for unknown roles
throw new ForbiddenException('تم رفض الوصول'); // Access denied
}

userId: string,
userRole: string
): Promise<any> {
// First verify access
await this.verifyPatientAccess(patientId, userId, userRole);
// Get patient data with role-based includes
const patient = await this.prisma.patient.findUnique({
where: { id: patientId },
include: {
// Appointments - patients see fewer, staff see more
appointments: {
orderBy: { dateTime: 'desc' },
take: userRole === 'PATIENT' ? 5 : 20,
include: {
therapist: {
select: {
id: true,
name: true,
},
},
},
},

// Invoices - patients only see paid ones
invoices: userRole === 'PATIENT' ? {
where: { status: 'PAID' },
orderBy: { createdAt: 'desc' },
} : {
orderBy: { createdAt: 'desc' },
},

// Medical files - patients see limited info
files: userRole === 'PATIENT' ? {
select: {
id: true,
filename: true,
mimeType: true,
createdAt: true,
},
} : true,

// Include based on role
...(userRole !== 'PATIENT' && {
// Staff can see user account info
user: {
select: {
id: true,
name: true,
phone: true,
email: true,
role: true,
},
},
}),
},
});

if (!patient) {
throw new NotFoundException('المريض غير موجود');
}

// Filter sensitive data for patients
if (userRole === 'PATIENT') {
// Remove internal notes that patients shouldn't see
delete patient.medicalHistory;

// Remove any internal staff notes
if (patient.appointments) {
patient.appointments = patient.appointments.map(apt => ({
...apt,
notes: apt.notes ? 'متاح للعرض' : null, // Available for viewing
}));
}
}

return patient;
}

/**

Filter list queries based on user role
Use this in list endpoints to automatically filter data
@param baseWhere - The base where clause from the query
@param userId - The current user's ID
@param userRole - The current user's role
@returns Modified where clause with role-based filtering
*/
buildFilteredWhere(
baseWhere: any = {},
userId: string,
userRole: string
): any {
// PATIENT: Only see their own data
if (userRole === 'PATIENT') {
return {
...baseWhere,
patient: {
userId,
},
};
}
// THERAPIST: See patients they have treated
if (userRole === 'THERAPIST') {
return {
...baseWhere,
OR: [
{
// Data from their appointments
appointment: {
therapistId: userId,
},
},
{
// Or direct patient relationship
patient: {
userId,
},
},
],
};
}

// OWNER and SECRETARY: See all data
return baseWhere;
}

/**

Check if user can modify patient data
@param patientId - The patient ID
@param userId - The current user's ID
@param userRole - The current user's role
@throws ForbiddenException if modification is denied
*/
async verifyPatientModificationAccess(
patientId: string,
userId: string,
userRole: string
): Promise<void> {
// Only OWNER, THERAPIST, and SECRETARY can modify patient data
if (!['OWNER', 'THERAPIST', 'SECRETARY'].includes(userRole)) {
throw new ForbiddenException(
'لا يمكنك تعديل بيانات المرضى' // You cannot modify patient data
);
}
// THERAPIST can only modify their own patients' basic info
if (userRole === 'THERAPIST') {
await this.verifyPatientAccess(patientId, userId, userRole);
}
}
